{{- $dest := .Destination -}} {{- $is_anchor := strings.HasPrefix $dest "#" -}} {{- $is_remote := or (strings.HasPrefix $dest "http") (strings.HasPrefix $dest "//") -}} {{- /* LOGIQUE : 1. Si c'est un lien web (http) ou une ancre (#top), on affiche direct. 2. Sinon, on essaie de trouver la page Hugo correspondante avec .GetPage (Liens relatifs) */ -}} {{- if or $is_remote $is_anchor -}} {{ .Text | safeHTML }} {{- else -}} {{- /* Tentative de résolution interne */ -}} {{- $page := .Page.GetPage $dest -}} {{- if $page -}} {{- /* CAS A : Page trouvée ! On utilise son permalien canonique */ -}} {{ .Text | safeHTML }} {{- else -}} {{- /* CAS B : Pas trouvé (Fichier statique, PDF, ou lien cassé). On laisse tel quel. */ -}} {{ .Text | safeHTML }} {{- end -}} {{- end -}}
