{{- $dest := .Destination -}} {{- $is_anchor := strings.HasPrefix $dest "#" -}} {{- $is_remote := or (strings.HasPrefix $dest "http") (strings.HasPrefix $dest "//") -}} {{- $is_scheme := or (strings.HasPrefix $dest "mailto:") (strings.HasPrefix $dest "tel:") -}} {{- /* Système de liens internes « hugoesque » : on écrit dans le markdown un chemin de contenu absolu ([texte](/section/page/#ancre)), et ce hook le résout via GetPage vers le permalien canonique. C'est le remplaçant moderne du shortcode relref : il dé-accentue (removePathAccents), survit aux changements de config de permaliens, et signale les liens cassés (classe broken-link) au lieu de les laisser passer silencieusement. Filet de sécurité relref : Hugo remplace les shortcodes par un marqueur opaque (HAHAHUGOSHORTCODE…HBHB) AVANT goldmark, donc le hook voit le marqueur, jamais l'URL. Un marqueur vient forcément d'un relref validé au build (refLinksErrorLevel = ERROR) : on le classe interne et on laisse Hugo réinjecter le href après coup. */ -}} {{- $is_shortcode := and (strings.Contains $dest "HUGOSHORTCODE") (strings.Contains $dest "HBHB") -}} {{- if or $is_remote $is_anchor $is_scheme -}} {{ .Text | safeHTML }} {{- else if $is_shortcode -}} {{ .Text | safeHTML }} {{- else -}} {{- /* urls.Parse sépare proprement chemin et fragment (#ancre) */ -}} {{- $parsed := urls.Parse $dest -}} {{- $path := $parsed.Path -}} {{- $fragment := "" -}} {{- with $parsed.Fragment -}}{{- $fragment = printf "#%s" . -}}{{- end -}} {{- $page := .Page.GetPage $path -}} {{- if $page -}} {{- /* Page trouvée : permalien canonique (accents retirés, config-proof) + ancre */ -}} {{ .Text | safeHTML }} {{- else if strings.HasPrefix $path "/" -}} {{- /* Chemin absolu non résolu par GetPage : sans doute une ressource statique ou un endpoint JSON (/antenne-radio/, fichiers servés) — on laisse passer tel quel. */ -}} {{ .Text | safeHTML }} {{- else -}} {{- /* Chemin relatif non résolu : vrai lien cassé, on le signale */ -}} {{ .Text | safeHTML }} {{- end -}} {{- end -}}
