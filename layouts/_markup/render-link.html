{{- $dest := .Destination -}} {{- $is_remote := (strings.HasPrefix $dest "http") -}} {{- $is_anchor := (strings.HasPrefix $dest "#") -}} {{- /* ================================================================================ CAS 1 : LIEN EXTERNE ou ANCRE Si le lien commence par "http" ou "#", on ne fait rien de spécial. ================================================================================ */ -}} {{- if or $is_remote $is_anchor -}} {{ .Text | safeHTML }} {{- /* ================================================================================ CAS 2 : LIEN INTERNE (ce qui nous intéresse) On va chercher la page par son nom de fichier. ================================================================================ */ -}} {{- else -}} {{- $found_page := "" -}} {{- $link_filename := $dest | path.Base -}} {{/* Prend "foo/bar/page.md" et garde "page.md" */}} {{- /* On ne lance la recherche que si le lien se termine par .md */ -}} {{- if strings.HasSuffix $link_filename ".md" -}} {{- /* C'est la magie : on boucle sur TOUTES les pages du site. C'est lent sur un site de 10 000 pages, mais parfait pour un blog. */ -}} {{- range site.RegularPages -}} {{- /* On compare le nom de fichier du lien (page.md) avec le nom de fichier de la page (.File.LogicalName) */ -}} {{- if eq .File.LogicalName $link_filename -}} {{- $found_page = . -}} {{/* On l'a trouvée ! */}} {{- end -}} {{- end -}} {{- end -}} {{- /* ============================================================================== RÉSULTAT : On a trouvé la page ! On utilise le lien permanent (.RelPermalink) de la page trouvée. ============================================================================== */ -}} {{- if $found_page -}} {{ .Text | safeHTML }} {{- /* ============================================================================== RÉSULTAT : Pas trouvé (c'est un lien cassé ou un lien vers une image /static/...) On laisse le lien tel quel. ============================================================================== */ -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}} {{- end -}}
